## Updated Solution: Responsive Sidebar with AppShell.Navbar

**Key behavior:**
- **Desktop (>sm breakpoint):** Sidebar always visible, clicking nav items does NOT close sidebar
- **Mobile (<=sm breakpoint):** Sidebar hidden, Burger opens Drawer, clicking nav items closes Drawer

**Implementation approach:**
- Use `AppShell.Navbar` with `withBorder` and `p="md"` for desktop sidebar
- Use `Drawer` component for mobile navigation overlay
- Use Mantine's `useMantineTheme` or media query to detect breakpoints
- Footer visibility depends ONLY on route (`/main/dashboard`), not sidebar state

```typescript
// MainLayout.tsx - Updated logic
import { useState } from "react";
import { 
  AppShell, 
  Burger, 
  Group, 
  Text, 
  Drawer, 
  NavLink, 
  Stack, 
  Button,
  useMantineTheme 
} from "@mantine/core";
import { useMediaQuery } from "@mantine/hooks";
import { Outlet, useLocation, useNavigate } from "react-router-dom";
import { sidebarItems, footerItems } from "./model/sidebarItems";

export function MainLayout() {
  const [drawerOpened, setDrawerOpened] = useState(false);
  const location = useLocation();
  const navigate = useNavigate();
  const isMobile = useMediaQuery("(max-width: 48em)");
  const showFooter = location.pathname.startsWith("/main/dashboard");

  const handleNavClick = (path: string) => {
    navigate(path);
    if (isMobile) {
      setDrawerOpened(false);
    }
  };

  return (
    <AppShell header={{ height: 60 }} padding="md">
      <AppShell.Header>
        <Group h="100%" px="md">
          {isMobile && (
            <Burger 
              opened={drawerOpened} 
              onClick={() => setDrawerOpened(!drawerOpened)} 
              size="sm" 
            />
          )}
          <Text fw={500}>My App</Text>
        </Group>
      </AppShell.Header>

      {/* Desktop Sidebar - always visible */}
      {!isMobile && (
        <AppShell.Navbar p="md" withBorder>
          <Stack>
            {sidebarItems.map((item) => (
              <NavLink
                key={item.path}
                label={item.label}
                leftSection={item.icon}
                active={location.pathname === item.path}
                onClick={() => handleNavClick(item.path)}
                style={{ borderRadius: "var(--mantine-radius-md)" }}
              />
            ))}
          </Stack>
        </AppShell.Navbar>
      )}

      {/* Mobile Drawer */}
      {isMobile && (
        <Drawer
          opened={drawerOpened}
          onClose={() => setDrawerOpened(false)}
          title="Menu"
          padding="md"
          size="xs"
        >
          <Stack>
            {sidebarItems.map((item) => (
              <NavLink
                key={item.path}
                label={item.label}
                leftSection={item.icon}
                active={location.pathname === item.path}
                onClick={() => handleNavClick(item.path)}
                style={{ borderRadius: "var(--mantine-radius-md)" }}
              />
            ))}
          </Stack>
        </Drawer>
      )}

      {/* Footer - visible only on Dashboard */}
      {showFooter && (
        <AppShell.Footer p="md">
          <Group gap="sm">
            {footerItems.map((item) => (
              <Button
                key={item.path}
                variant="light"
                size="sm"
                onClick={() => navigate(item.path)}
              >
                {item.label}
              </Button>
            ))}
          </Group>
        </AppShell.Footer>
      )}

      <AppShell.Main>
        <Outlet />
      </AppShell.Main>
    </AppShell>
  );
}
```

```typescript
// MainLayout.tsx - Final Implementation
import { useState } from "react";
import { 
  AppShell, 
  Burger, 
  Group, 
  Text, 
  Drawer, 
  NavLink, 
  Stack, 
  Button,
  ActionIcon 
} from "@mantine/core";
import { useMediaQuery } from "@mantine/hooks";
import { Outlet, useLocation, useNavigate } from "react-router-dom";
import { sidebarItems, footerItems } from "./model/sidebarItems";
import { IconX } from "@tabler/icons-react";

export function MainLayout() {
  const [drawerOpened, setDrawerOpened] = useState(false);
  const [sidebarOpened, setSidebarOpened] = useState(true); // For desktop sidebar toggle
  const location = useLocation();
  const navigate = useNavigate();
  const isMobile = useMediaQuery("(max-width: 48em)");
  const showFooter = location.pathname.startsWith("/main/dashboard");

  const handleNavClick = (path: string) => {
    navigate(path);
    if (isMobile) {
      setDrawerOpened(false);
    }
  };

  return (
    <AppShell
      header={{ height: 60 }}
      padding="md"
      footer={showFooter ? <AppShell.Footer p="md">
        <Group gap="sm">
          {footerItems.map((item) => (
            <Button
              key={item.path}
              variant="light"
              size="sm"
              onClick={() => navigate(item.path)}
            >
              {item.label}
            </Button>
          ))}
        </Group>
      </AppShell.Footer> : undefined}
    >
      <AppShell.Header>
        <Group h="100%" px="md">
          {(isMobile || !sidebarOpened) && (
            <Burger
              opened={isMobile ? drawerOpened : !sidebarOpened}
              onClick={() => isMobile ? setDrawerOpened(!drawerOpened) : setSidebarOpened(!sidebarOpened)}
              size="sm"
            />
          )}
          <Text fw={500}>My App</Text>
        </Group>
      </AppShell.Header>

      {/* Desktop Sidebar - toggleable */}
      {!isMobile && sidebarOpened && (
        <AppShell.Navbar p="md" withBorder>
          <Group justify="flex-end" mb="md">
            <ActionIcon variant="subtle" onClick={() => setSidebarOpened(false)}>
              <IconX size={18} />
            </ActionIcon>
          </Group>
          <Stack>
            {sidebarItems.map((item) => (
              <NavLink
                key={item.path}
                label={item.label}
                leftSection={item.icon}
                active={location.pathname === item.path}
                onClick={() => handleNavClick(item.path)}
                style={{ borderRadius: "var(--mantine-radius-md)" }}
              />
            ))}
          </Stack>
        </AppShell.Navbar>
      )}

      {/* Mobile Drawer */}
      {isMobile && (
        <Drawer
          opened={drawerOpened}
          onClose={() => setDrawerOpened(false)}
          title="Menu"
          padding="md"
          size="xs"
        >
          <Stack>
            {sidebarItems.map((item) => (
              <NavLink
                key={item.path}
                label={item.label}
                leftSection={item.icon}
                active={location.pathname === item.path}
                onClick={() => handleNavClick(item.path)}
                style={{ borderRadius: "var(--mantine-radius-md)" }}
              />
            ))}
          </Stack>
        </Drawer>
      )}

      <AppShell.Main>
        <Outlet />
      </AppShell.Main>
    </AppShell>
  );
}
```

## Critical Fixes Required

### Issue 1: Burger icon when sidebar is closed
- **Problem:** When sidebar is closed on desktop, user sees X (close) icon instead of burger (open) icon
- **Solution:** Always show Burger icon in Header. When sidebar is open → Burger shows X. When sidebar is closed → Burger shows bars (open icon).

### Issue 2: Close button position
- **Problem:** X button is inside Navbar, not near My App header
- **Solution:** Burger in Header controls sidebar open/close state. No separate X button needed in navbar.

### Issue 3: Footer belongs to DashboardPage, NOT MainLayout
- **Problem:** Footer is in MainLayout, but user wants DashboardPage to have its own Footer
- **Solution:** 
  - Remove Footer from MainLayout completely
  - DashboardPage wraps content in `AppShell` with `AppShell.Footer`
  - MainLayout provides ONLY Header + Navbar structure
